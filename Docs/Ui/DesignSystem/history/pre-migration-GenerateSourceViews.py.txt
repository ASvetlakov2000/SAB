def replace_method(text,signature,body):
    start=text.index(signature); opening=text.index('{',start)
    mask=re.sub(r'@"(?:""|[^"])*"|"(?:\\.|[^"\\])*"|\'(?:\\.|[^\'\\])*\'|//[^\n]*|/\*[\s\S]*?\*/',lambda m:' '*(m.end()-m.start()),text)
    depth=1;i=opening+1
    while depth:
        if mask[i]=='{':depth+=1
        elif mask[i]=='}':depth-=1
        i+=1
    return text[:opening]+'{\n'+body+'\n        }'+text[i:]
def method_body(text,signature):
    start=text.index(signature);opening=text.index('{',start)
    mask=re.sub(r'@"(?:""|[^"])*"|"(?:\\.|[^"\\])*"|\'(?:\\.|[^\'\\])*\'|//[^\n]*|/\*[\s\S]*?\*/',lambda m:' '*(m.end()-m.start()),text)
    depth=1;i=opening+1
    while depth:
        if mask[i]=='{':depth+=1
        elif mask[i]=='}':depth-=1
        i+=1
    return text[opening:i]
